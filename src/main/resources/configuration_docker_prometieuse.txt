# Sur Mac
cd ~/Desktop
mkdir prometheus
cd prometheus
nano prometheus.yml

global:
  scrape_interval: 15s
  evaluation_interval: 15s

scrape_configs:
  - job_name: 'user-management'
    metrics_path: '/actuator/prometheus'
    static_configs:
      - targets: ['host.docker.internal:8080']
lancer

docker run -d --name prometheus -p 9090:9090 -v "C:\FormationSpringData\user-management\prometheus\prometheus.yml:/etc/prometheus/prometheus.yml" prom/prometheus
